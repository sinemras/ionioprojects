
#include<stdio.h>
int main(){
	int num1,num2,sum,dif,mul;
	scanf("%d %d",&num1,&num2);
	sum = num1 + num2;
	dif = num1 - num2;
	mul = num1 * num2;
	if(num2 != 0){
	       float div = (float)num1 / num2;
		printf("  sum: %d \n  difference: %d \n  multiple: %d \n  division: %f \n",sum,dif,mul,div);
	}
	else{
		printf("  sum: %d \n  difference: %d \n  multiple: %d \n  division: undefined \n",sum,dif,mul);
	}
	 return 0;
}
